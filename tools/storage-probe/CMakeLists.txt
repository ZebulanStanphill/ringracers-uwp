cmake_minimum_required(VERSION 3.14)
set(CMAKE_SYSTEM_NAME WindowsStore)
set(CMAKE_SYSTEM_VERSION 10.0.19041.0)
project(storage-probe LANGUAGES CXX)
add_executable(storage-probe WIN32 main.cpp Package.appxmanifest)
target_compile_features(storage-probe PRIVATE cxx_std_17)
target_compile_definitions(storage-probe PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(storage-probe PRIVATE /EHsc /W4)
target_link_libraries(storage-probe PRIVATE WindowsApp.lib d3d11.lib dxgi.lib)
set_property(TARGET storage-probe PROPERTY VS_WINDOWS_TARGET_PLATFORM_MIN_VERSION 10.0.19041.0)
set(assets Assets/StoreLogo.png Assets/Square44x44Logo.png Assets/Square150x150Logo.png)
set_source_files_properties(${assets} PROPERTIES VS_DEPLOYMENT_CONTENT TRUE VS_DEPLOYMENT_LOCATION Assets)
target_sources(storage-probe PRIVATE ${assets})
